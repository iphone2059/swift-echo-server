cmake_minimum_required(VERSION 3.24)
project(swift_echo_server_verification LANGUAGES NONE)
include(CTest)
if(NOT SWIFT_CONFIGURATION MATCHES "^(debug|release)$" OR NOT IS_DIRECTORY "${SWIFT_BIN_PATH}")
  message(FATAL_ERROR "Pass the built Swift configuration and product directory")
endif()
find_program(PWSH_EXECUTABLE NAMES pwsh REQUIRED)
function(swift_gate name script)
  add_test(NAME "${name}" COMMAND "${PWSH_EXECUTABLE}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/Tests/${script}" ${ARGN})
  set_tests_properties("${name}" PROPERTIES TIMEOUT 300 RUN_SERIAL TRUE WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endfunction()
swift_gate(swift-unit ces_unit_tests.ps1 -Configuration "${SWIFT_CONFIGURATION}")
swift_gate(source-policy ces_source_policy.ps1 )
swift_gate(ownership-features ces_ownership_compile_tests.ps1 -Configuration "${SWIFT_CONFIGURATION}" -Phase features)
swift_gate(ownership-owners ces_ownership_compile_tests.ps1 -Configuration "${SWIFT_CONFIGURATION}" -Phase owners)
swift_gate(memory-safety ces_memory_safety_checks.ps1 -BinPath "${SWIFT_BIN_PATH}" -Configuration "${SWIFT_CONFIGURATION}")
swift_gate(gate-regressions ces_verification_gate_tests.ps1 -Configuration "${SWIFT_CONFIGURATION}")
swift_gate(fault-processes ces_fault_process_tests.ps1 -DriverPath "${SWIFT_BIN_PATH}/swift-echo-server-fault-driver.exe")
swift_gate(process-contract ces_process_tests.ps1 -ServerPath "${SWIFT_BIN_PATH}/swift-echo-server.exe")
swift_gate(extended-process-contract ces_extended_process_tests.ps1 -ServerPath "${SWIFT_BIN_PATH}/swift-echo-server.exe")
swift_gate(cross-implementation ces_interop_tests.ps1 -ServerPath "${SWIFT_BIN_PATH}/swift-echo-server.exe" -Configuration "${SWIFT_CONFIGURATION}")
swift_gate(reset-storm ces_reset_storm_tests.ps1 -ServerPath "${SWIFT_BIN_PATH}/swift-echo-server.exe" -Label swift-echo-server)
